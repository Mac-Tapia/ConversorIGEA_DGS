# Conversor IGEA/CYMDIST → DGS — dependencias de producción
#
#   python -m venv .venv
#   .venv\Scripts\activate
#   python -m pip install -r requirements.txt
#
# Instala TODO lo que la interfaz necesita para funcionar completa: las dos
# alternativas de entrada (TXT y base Access), la georreferenciación, las plantillas
# Excel de cargas y la validación independiente del modelo.
#
# Para pruebas y desarrollo:  pip install -r requirements-dev.txt
# Python >= 3.10. En Linux la GUI suele necesitar el paquete del sistema python3-tk.
#
# Nota sobre PowerFactory: su API (powerfactory.pyd) NO se instala con pip. Viene con
# DIgSILENT y está atada a una versión concreta de CPython (PF 2024 → 3.12). El
# conversor la usa desde un proceso aparte; ver docs/MANUAL_COMANDOS.md §5.

# Paquete del proyecto (src/igea_dgs + schemas/*.json)
-e .

# Tercera alternativa de entrada: paquete oficial VNR-GIS. Mantener esta lista en
# un archivo propio permite instalar solo el adaptador en integraciones externas,
# pero la interfaz principal la incluye para que sus tres modos estén operativos.
-r vnr_requirements.txt

# Georreferenciación: GPS en terminales y diagrama IntGrf.
# Sin esto solo se puede convertir con --no-geography.
pyproj>=3.6

# Alternativa 2 de entrada: base de datos Access de CYMDIST (.mdb).
# Solo Windows, y requiere además el driver «Microsoft Access Driver (*.mdb, *.accdb)»
# de 64 bits, que se instala aparte con Microsoft Access Database Engine.
pyodbc>=5.0 ; sys_platform == "win32"

# Plantillas de cargas de SED en Excel (una hoja por alimentador) y export .xlsx.
openpyxl>=3.1

# Tablas de las exportaciones analíticas.
pandas>=2.0

# Validación independiente del modelo: flujo de potencia y detección de islas con un
# motor distinto de PowerFactory, sin licencia y en segundos. Ver docs del diagnóstico.
pandapower>=3.0

# Interfaz web (la interfaz principal): API FastAPI + front React compilado.
# El front se compila una vez con Node (cd frontend && npm install && npm run build);
# el resultado, frontend/dist, lo sirve esta misma API.
fastapi>=0.115
uvicorn[standard]>=0.30
python-multipart>=0.0.9
